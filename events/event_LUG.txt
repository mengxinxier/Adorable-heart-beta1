	id = LUG.1
	title = LUG.1.t
	desc = LUG.1.d
	picture = GFX_report_event_soviet_army_greeted
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = LUG.1.a
		activate_mission = GRE_witch_cult_purge
		hidden_effect = {
			kill_country_leader = yes
		}
		LUG_Emilia = {
			add_country_leader_role = {
				country_leader = {
					desc = "LUG_Emilia_desc"
					ideology = Absolute_monarchy
					traits = {
						LUG_traits_emt
					}
					expire = "1999.1.1.1"
					id = -1
				}
				promote_leader = yes
			}
		}
		set_power_balance = {
			id = LUG_Line_choose_bop
			left_side = LUG_Line_choose_bop_left_side
			right_side = LUG_Line_choose_bop_right_side
			set_value = 0.1
		}
		add_ideas = LUG_anti_half_elf
		add_ideas = LUG_Subaru_idea
		remove_ideas = LUG_illness
		clr_country_flag = no_emt
	}
}
