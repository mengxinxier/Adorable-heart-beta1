#486！满血复活
country_event = {
	id = LUG.8
	title = LUG.8.t
	desc = LUG.8.d
	picture = GFX_LUG_remu
	fire_only_once = no
	trigger = {
		tag = GRE
		AND = {
            check_variable = { var = Subaru_mental_state value = 0 compare = greater_than_or_equals }
            check_variable = { var = Subaru_mental_state value = 20 compare = less_than }
        }
	}
	option = {
		name = LUG.8.a
		set_variable = { var = Subaru_mental_state value = 100 }
		custom_effect_tooltip = LUG.8.a.desc
		LUG_Subaru_mental_state_check_effect = yes
		clamp_variable = { var = Subaru_mental_state min = 0 max = 100 }
	}
}
