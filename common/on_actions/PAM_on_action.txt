on_actions = {
	on_startup = {
		effect = {
			every_state = {
				limit = { is_core_of = PAM }
				set_state_flag = PAM_orgin_core_state
			}

        }
    }
    on_declare_war = {
        effect = {
            #南美大围猎
            if = {
                limit = {
                    ROOT = {
                        has_completed_focus = PAM_lqslzs
                        or = {
                            has_war_together_with = PAM
                            tag = PAM
                            is_in_faction_with = PAM
                            is_subject_of = PAM
                        }
                    }
                    FROM = {
                        or = {
                            tag = PAO
                            is_in_faction_with = PAO
                            tag = PAQ
                            is_in_faction_with = PAQ
                            tag = MGU
                            is_in_faction_with = ELF
                        }
                    }
                    not = {
                        has_global_flag = LHS_nmgm
                    }
                }
                hidden_effect = {
                    set_global_flag = LHS_nmgm
                    news_event = {
                        id = PAM_news.1
                        days = 0
                    }
                }
            }

            #最后一餐
             if = {
                 limit = {
                     ROOT = {
                         has_completed_focus = PAM_devour_the_old_world
                         or = {
                             has_war_together_with = PAM
                             tag = PAM
                             is_in_faction_with = PAM
                             is_subject_of = PAM
                         }
                     }
                     FROM = {
                         capital_scope = { is_on_continent = europe }
                     }
                     not = {
                         has_global_flag = PAM_last_war
                     }
                 }
                 hidden_effect = {
                     set_global_flag = PAM_last_war
                     news_event = {
                         id = PAM_news.2
                         days = 0
                     }
                 }
             }
        }
    }
    on_annex = {
        effect = {
            if = {
                limit = { #野火小果子线时击败亡灵会同步吞并无政府地区
                    ROOT = {
                        tag = PAM
                        has_war_with = LHS
                    }
                    FROM = {
                        tag = ZDW
                    }
                    LHS = {
                        is_ai = yes
                    }
                }
                PAM = {
                    annex_country = {
                        target = WRQ
                        transfer_troops = no
                    }
                }
            }
            if = {
                #战胜亡灵
                limit = {
                    PAM = {
                        NOT = {
                            has_country_flag = PAM_cishawldn #野火ai小果子线时不会触发之后的事件
                        }
                    }
                    ROOT = {
                        PAM = {has_completed_focus = PAM_cztm}
                        PAM = {
                            not = {
                                has_completed_focus = PAM_lqslzs
                            }
                        }
                        not = {has_country_flag = PAM_zswl}
                    }
                    FROM = {
                        tag = ZDW
                    }
                }
                PAM = {
                    set_country_flag = PAM_zswl
                    country_event = {
                        hours = 6
                        id = PAM.13
                    }
                }
            }
            if = {
                #战胜魔法少女联盟
                limit = {
                    ROOT = {
                        tag = PAM
                        has_completed_focus = PAM_zsblgz
                    }
                    FROM = {
                        tag = MGU
                    }
                }
                PAM = {
                    country_event = {
                        hours = 6
                        id = PAM.30
                    }
                }
            }
            if = {
                #吞并夜莺后清零亡灵装备
                limit = {
                    ROOT = {
                        tag = PAM
                    }
                    FROM = {
                        tag = PAO
                    }
                }
                PAM = {
                   add_equipment_to_stockpile = {
                       type = wangling_equipment
                       amount = -1500000
                   }
                }
            }
        }
    }
    on_capitulation = {
        effect = {
            if = {
                #战胜躁动的亡者
                limit = {
                    ROOT = {
                        tag = ZDW
                        is_subject = no
                        NOT = {
                            any_enemy_country = {
                                NOT = {
                                    tag = PAM
                                    has_war_together_with = PAM
                                }
                            }
                        }
                    }
                    FROM = {
                        NOT = {has_country_flag = PAM_zswl2}
                        or = {
                            tag = PAO
                        }
                    }
                }
                ZDW = {
                   add_equipment_to_stockpile = {
                       type = wangling_equipment
                       amount = -1500000
                   }
                }
                PAO = {
                    add_equipment_to_stockpile = {
                        type = wangling_equipment
                        amount = -1500000
                    }
                }
                PAM = {
                    set_country_flag = PAM_zswl2
                }
            }

        }
    }

}
