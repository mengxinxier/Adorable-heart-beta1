on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					or = {
						tag = PAW
					}
				}
				set_variable = {
					another_country_leader_group = 12
				}
				activate_mission = PAW_shijiezhiwai_gc_1
			}
		}
	}
	on_startup = {
		effect = {
			every_country = {
				limit = {
					tag = PAW
					is_ai = yes
				}
				country_event = paw.0
			}
		}
	}
	on_startup = {
		effect = {
			every_country = {
				limit = {
					tag = PAW
					is_ai = no
				}
				country_event = paw.10001
			}
		}
	}
	on_daily_PAW = { 
		effect = {
			if = {
				limit = { 
					tag = PAW
					date > 1936.6.1
					not = { has_country_flag = kelv_buhuisi }
					not = { has_country_flag = kelv_yuci }
				}
				set_country_flag = kelv_yuci
				country_event = paw.1
			}
		}
		effect = {
			if = {
				limit = { 
					tag = PAW
					date > 1937.6.1
					has_country_flag = lunhui_diyici_yes
					not = { has_country_flag = kelv_buhuisi }
					not = { has_country_flag = kelv_yuci }
				}
				set_country_flag = kelv_yuci
				country_event = paw.1
			}
		}
		effect = {
			if = {
				limit = { 
					tag = PAW
					date > 1938.6.1
					has_country_flag = lunhui_diyici_yes
					has_country_flag = lunhui_dierci_yes
					not = { has_country_flag = kelv_buhuisi }
					not = { has_country_flag = kelv_yuci }
				}
				set_country_flag = kelv_yuci
				country_event = paw.1
			}
		}
	}
}
