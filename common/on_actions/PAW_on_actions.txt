on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					or = {
						tag = PAW
					}
				}
				set_variable = {
					another_country_leader_group = 12
				}
				activate_mission = PAW_shijiezhiwai_gc_1
			}
			every_country = {
				limit = {
					tag = PAW
					is_ai = yes
				}
				country_event = paw.0
			}
			every_country = {
				limit = {
					tag = PAW
					is_ai = no
				}
				country_event = paw.10001
			}
		}
	}
	on_capitulation = {
		effect = {	
			if = {	#停止抵抗后计算国家数-1
				FROM = {
					limit = {
						has_global_flag = paw_heihualuxian_yes
						has_country_flag = paw_blue_and_red_tag
						is_in_array = { global.paw_msze_dysz_modifier = PREV }
					}
					clr_country_flag = paw_blue_and_red_tag
					remove_from_array = { array = global.paw_msze_dysz_modifier value = FROM }
					add_to_variable = {
						var = global.paw_mszesz_dysz_modifier
						value = -1
					}
				}
			}
		}
	}
	on_annex = {
		effect = {
			if = {	#被吞并后计算国家数-1
				FROM = {
					limit = {
						has_global_flag = paw_heihualuxian_yes
						has_country_flag = paw_blue_and_red_tag
						is_in_array = { global.paw_msze_dysz_modifier = PREV }
					}
					clr_country_flag = paw_blue_and_red_tag
					remove_from_array = { array = global.paw_msze_dysz_modifier value = FROM }
					add_to_variable = {
						var = global.paw_mszesz_dysz_modifier
						value = -1
					}
				}
			}
		}
		
	}
}
