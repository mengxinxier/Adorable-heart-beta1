scripted_gui = {
	Red_Ledger_window = {
		window_name = "Red_Ledger_window"
		visible = {
			tag = LHS
			has_country_flag = Red_Ledger_window_visible
		}
		effects = {
			close_button_click = {
				if = {
					limit = {
						has_country_flag = Red_Ledger_window_visible
					}
					clr_country_flag = Red_Ledger_window_visible
				}
				else = {
					set_country_flag = Red_Ledger_window_visible
				}
			}
		}
	}
	Red_Ledger_open_button_window = {
		window_name = "Red_Ledger_open_button_window"
		parent_window_token = politics_tab
		visible = {
			tag = LHS
			always = no
		}
		properties = {
		}
		effects = {
			Red_Ledger_open_button_click = {
				if = {
					limit = {
						has_country_flag = Red_Ledger_window_visible
					}
					clr_country_flag = Red_Ledger_window_visible
				}
				else = {
					set_country_flag = Red_Ledger_window_visible
				}
				if = {
					limit = {
						not = { has_country_flag = Red_Ledger_open_arrows_alert_unvisible }
					}
					set_country_flag = Red_Ledger_open_arrows_alert_unvisible
				}
			}
		}
	}
	Red_Ledger_open_arrows_alert = {
		context_type = player_context
		parent_window_token = politics_tab
		window_name = "Red_Ledger_open_arrows_alert"
		visible = {
			tag = LHS
			always = no
			not = { has_country_flag = Red_Ledger_open_arrows_alert_unvisible }
		}
		properties = {
		}
		effects = {
		}
	}
}
