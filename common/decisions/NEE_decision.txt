	CAN_NEE_start_integration_efforts = {
		icon = generic_prepare_civil_war
		allowed = { tag = CAN }
		available = {
			CAN = { has_completed_focus = CAN_The_Halifax_talks }

		visible = {

		}

		cost = 0
		
		ai_will_do = { factor = 100 }
		
		complete_effect = {
			set_country_flag = CAN_NEE_integration_started
			
			# Activate the loop mechanics
			set_country_flag = CAN_NEE_interference_active 
			
			# Trigger the main crisis timer (handled by availability of the mission below)
			# Start the clock log
			log = "[GetDateText]: CAN Integration of NEE Started."

	CAN_NEE_independence_crisis = {
		icon = generic_disaster_icon
		allowed = { tag = CAN }
		
		# Auto-active once integration starts
		activation = {
			has_country_flag = CAN_NEE_integration_started
			NEE = { exists = yes }
		}
		available = {
			NEE = { exists = yes }
		}
		
		days_mission_timeout = 540
		is_good = no
		selectable_mission = no # Runs automatically
		
		timeout_effect = {
			CAN = {
				log = "[GetDateText]: Mission Failed - NEE Independence"
				clr_country_flag = CAN_NEE_integration_started
				clr_country_flag = CAN_NEE_interference_active
			}
			NEE = {
				country_event = { id = nee_unification.2 }

			}

	# Simulates the struggle. Spawns periodically.
	# Failure (-10 Influence) represents losing ground.
	CAN_NEE_british_interference = {
		icon = generic_spy_intel #[Checks if exists? generic_spy_intel usually ok, or generic_counter_intelligence]
		allowed = { tag = CAN }
		
		# Controlled by flag
		activation = {
			has_country_flag = CAN_NEE_interference_active
			has_country_flag = CAN_NEE_integration_started
			NEE = { exists = yes }
		}
		
		# 90 Days to respond
		days_mission_timeout = 90
		is_good = no
		selectable_mission = no # Auto-trigger based on flag
		
		# Timeout = Failure to contain
		timeout_effect = {
			add_to_variable = { var = can_nee_influence value = -10 }
			log = "[GetDateText]: British/NEE interference succeeded. Influence lost."
			
			# Disable mission temporarily
			clr_country_flag = CAN_NEE_interference_active
			
			# Re-enable after short delay (30 days) - Relentless pressure
			CAN = { 
				country_event = { 
					id = can_nee_internal.998 # Placeholder ID for logic 
					days = 30 
					random_days = 15
				} 
			}
		}
	}

	# 对抗抵抗决议 / COUNTER-INTERFERENCE DECISION
	# -------------------------------------------------------------------------
	# Pays PP to solve the active interference mission.
	CAN_NEE_counter_interference = {
		icon = generic_intel_bonus
		allowed = { tag = CAN }
		
		available = {
			is_mission_active = CAN_NEE_british_interference
		}
		
		cost = 25 # PP Cost to solve
		fire_only_once = no
		
		ai_will_do = {
			factor = 10
			modifier = {
				factor = 0
				pp < 50
			}
		}
		
		complete_effect = {
			# Bonus for solving it actively? Maybe small +2
			add_to_variable = { var = can_nee_influence value = 2 }
			log = "[GetDateText]: Interference countered by agent action."

			CAN = {

					id = can_nee_internal.998 
					days = 60
					random_days = 30
				} 
			}
		}
	}
