
KMK_KNM_WAR_descision = {
	KNM_unit_KMK_war = {
		icon = GFX_decision_hol_draw_up_staff_plans
		available = {
			hidden_trigger = {
				always = no
			}
		}
		priority = 20
		days_mission_timeout = 100
		fire_only_once = yes
		activation = {
			always = no
		}
		visible = {
		}
		cancel_if_not_visible = yes
		selectable_mission = yes
		cancel_trigger = {
			or = {
				KNM = {
					controls_state = 925
				}
				KMK = {
					controls_state = 7
				}
			}
		}
		cancel_effect = {
			if = {
				limit = {
					KNM = {
						controls_state = 925
					}
				}
				if = {
					limit = {
						KNM = {
							has_government = communism
						}
					}
					KNM = {
						white_peace = {
							tag = KMK
							message = KNM_WIN_KMK_c
						}
					}
				}
				else = {
					KNM = {
						white_peace = {
							tag = KMK
							message = KNM_WIN_KMK_n
						}
					}
				}
				KNM = {
					annex_country = {
						target = KMK
						transfer_troops = yes
					}
					set_temp_variable = { tag_to_release = LIB }
					puppet_released_tag_and_old_govern = yes
					set_temp_variable = { tag_to_release = COG }
					puppet_released_tag_and_old_govern = yes
				}
			}
			else = {
				if = {
					limit = {
						KNM = {
							has_government = communism
						}
					}
					KMK = {
						white_peace = {
							tag = KNM
							message = KMK_WIN_KNM_c
						}
					}
				}
				else = {
					KMK = {
						white_peace = {
							tag = KNM
							message = KMK_WIN_KNM_n
						}
					}
				}
				KMK = {
					annex_country = {
						target = KNM
						transfer_troops = yes
					}
				}
			}
		}
	}
}
