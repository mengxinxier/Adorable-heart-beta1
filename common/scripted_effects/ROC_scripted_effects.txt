ROC_policy_window_button_open_effect = {  ##打开UI
	clear_array = ROC_policy_select_card_Military_array
	clear_array = ROC_policy_select_card_Economy_array
	clear_array = ROC_policy_select_card_Diplomacy_array
	clear_array = ROC_policy_select_card_Generic_array
	clear_array = ROC_policy_unlock_card_array
	clear_array = ROC_policy_unlock_card_array_temp
	for_each_loop = {
		array = ROC_policy_select_card_Military_array_ture
		value = v
		break = break
		add_to_array = { ROC_policy_select_card_Military_array = v }
	}
	for_each_loop = {
		array = ROC_policy_select_card_Economy_array_ture
		value = v
		break = break
		add_to_array = { ROC_policy_select_card_Economy_array = v }
	}
	for_each_loop = {
		array = ROC_policy_select_card_Diplomacy_array_ture
		value = v
		break = break
		add_to_array = { ROC_policy_select_card_Diplomacy_array = v }
	}
	for_each_loop = {
		array = ROC_policy_select_card_Generic_array_ture
		value = v
		break = break
		add_to_array = { ROC_policy_select_card_Generic_array = v }
	}
	for_each_loop = {
		array = ROC_policy_unlock_card_array_ture
		value = v
		break = break
		add_to_array = { ROC_policy_unlock_card_array_temp = v }
	}
	while_loop_effect = {
		limit = {
			any_of = {
				array = ROC_policy_unlock_card_array_temp
				value = v
				index = i
				check_variable = { v < 999 }
			}
		}
		find_lowest_in_array = {
			array = ROC_policy_unlock_card_array_temp
			value = temp_lowest_v
			index = temp_i
		}
		add_to_array = { ROC_policy_unlock_card_array = temp_lowest_v }
		set_variable = { ROC_policy_unlock_card_array_temp^temp_i = 999 }
	}
}
