		find_lowest_in_array = {
			array = ROC_policy_unlock_card_array_temp
			value = temp_lowest_v
			index = temp_i
		}
		add_to_array = { ROC_policy_unlock_card_array = temp_lowest_v }

ROC_factional_influence_calculation = {  ##计算广府派系影响力
	set_variable = { ROC_total_influence_var = 0 }
	if = {
		limit = { is_in_array = { KMT_main_party_array = 2 } }
		add_to_variable = { ROC_total_influence_var = kmt_party_percent^2 }
	}
	if = {
		limit = { is_in_array = { KMT_main_party_array = 3 } }
		add_to_variable = { ROC_total_influence_var = kmt_party_percent^3 }
	}
	if = {
		limit = { is_in_array = { KMT_main_party_array = 4 } }
		add_to_variable = { ROC_total_influence_var = kmt_party_percent^4 }
	}
	if = {
		limit = { is_in_array = { KMT_main_party_array = 5 } }
		add_to_variable = { ROC_total_influence_var = kmt_party_percent^5 }
	}
	if = {
		limit = { is_in_array = { KMT_main_party_array = 6 } }
		add_to_variable = { ROC_total_influence_var = kmt_party_percent^6 }
	}
	if = {
		limit = { is_in_array = { KMT_main_party_array = 7 } }
		add_to_variable = { ROC_total_influence_var = kmt_party_percent^7 }
	}
	divide_variable = { kmt_party_percent^2 = ROC_total_influence_var }
	divide_variable = { kmt_party_percent^3 = ROC_total_influence_var }
	divide_variable = { kmt_party_percent^4 = ROC_total_influence_var }
	divide_variable = { kmt_party_percent^5 = ROC_total_influence_var }
	divide_variable = { kmt_party_percent^6 = ROC_total_influence_var }
	divide_variable = { kmt_party_percent^7 = ROC_total_influence_var }
	clear_array = kmt_party_percent_ture
	set_variable = { kmt_party_percent_ture^2 = kmt_party_percent^2 }  ##民革
	set_variable = { kmt_party_percent_ture^3 = kmt_party_percent^3 }  ##宪政会
	set_variable = { kmt_party_percent_ture^4 = kmt_party_percent^4 }  ##CC
	if = {
		limit = { check_variable = { kmt_party_overlord^5 = 2 } }  ##中正
		add_to_variable = { kmt_party_percent_ture^2 = kmt_party_percent^5 }
		set_variable = { kmt_party_percent_ture^5 = 0 }
	}
	else = { set_variable = { kmt_party_percent_ture^5 = kmt_party_percent^5 } }
	if = {
		limit = { check_variable = { kmt_party_overlord^6 = 2 } }  ##蓝衣社
		add_to_variable = { kmt_party_percent_ture^2 = kmt_party_percent^6 }
	}
	else = { add_to_variable = { kmt_party_percent_ture^5 = kmt_party_percent^6 } }
	if = {
		limit = { is_in_array = { KMT_main_party_array = 7 } }
		set_variable = { kmt_party_percent_ture^7 = kmt_party_percent^7 }  ##改组
	}
	else = { set_variable = { kmt_party_percent_ture^7 = 0 } }
}
