	LFC_yulin_xuanxiang_effect = { #雨林事件选项判定是否触发成功
		if = {
			limit = {
				check_variable = {
					LFC_VAR_Rainforest_supply < 0.1
				}
			}
			country_event = LFC_event_forest.4
		}
		else_if = {
			limit = {
				check_variable = {
					LFC_VAR_Rainforest_manpower < 0.1
				}
			}
			country_event = LFC_event_forest.5
		}
		else_if = {
			limit = {
				check_variable = {
					LFC_VAR_Rainforest_progress > 99
				}
			}
			country_event = LFC_event_forest.6
		}
		else = {
			LFC_Preset_yulin_event_effect = yes
		}

	}
