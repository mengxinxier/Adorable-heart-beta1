# === 计算基建加成 ===
# 每1点基建指数 = 0.2% 建造速度，0.2% 工厂槽位，-0.2% 补给消耗
LHS_calculate_infrastructure_bonus = {
    set_variable = { kapok_infrastructure_buildings_bonus = kapok_infrastructure }
    multiply_variable = { kapok_infrastructure_buildings_bonus = 0.002 }  # 1点+0.2%建造速度
    
    set_variable = { kapok_infrastructure_slots_bonus = kapok_infrastructure }
    multiply_variable = { kapok_infrastructure_slots_bonus = 0.002 }
    
    set_variable = { kapok_infrastructure_supply_bonus = kapok_infrastructure }
    multiply_variable = { kapok_infrastructure_supply_bonus = -0.002 }  # 负数=减少消耗
}

# === 计算福利加成===
LHS_calculate_welfare_bonus = {
    # Debuff部分
    set_variable = { kapok_welfare_consumer_goods_penalty = kapok_welfare }
    multiply_variable = { kapok_welfare_consumer_goods_penalty = 0.002 }  # 每1点+0.2%消费品
    
    # Buff部分（超过debuff的补偿）
    set_variable = { kapok_welfare_population_bonus = kapok_welfare }
    multiply_variable = { kapok_welfare_population_bonus = 0.005 }  # 每1点+0.005%每日顺从度增长
    
    set_variable = { kapok_welfare_stability_bonus = kapok_welfare }
    multiply_variable = { kapok_welfare_stability_bonus = 0.002 }  # 每1点+0.2%稳定
    
    set_variable = { kapok_welfare_efficiency_bonus = kapok_welfare }
    multiply_variable = { kapok_welfare_efficiency_bonus = 0.003 }  # 每1点+0.3%生产效率增长
}

# === 计算平衡奖励 ===
# 检测民工和军工的差距，差距越小奖励越大
LHS_calculate_balance_bonus = {
    # 计算失衡惩罚值
    set_variable = { temp_civ = num_of_civilian_factories }
    set_variable = { temp_mil = num_of_military_factories }
    
    # 如果民工>军工
    if = {
        limit = { 
            check_variable = { temp_civ > temp_mil }
        }
        set_variable = { kapok_balance_penalty = temp_civ }
        subtract_from_variable = { kapok_balance_penalty = temp_mil }
    }
    # 如果军工>民工
    else = {
        set_variable = { kapok_balance_penalty = temp_mil }
        subtract_from_variable = { kapok_balance_penalty = temp_civ }
    }
    
    # 计算奖励（差距越小，奖励越大）
    
    set_variable = { kapok_balance_political_power = 20 }  # 基础20
    subtract_from_variable = { kapok_balance_political_power = kapok_balance_penalty }
    if = {
        limit = { check_variable = { kapok_balance_political_power < 0 } }
        set_variable = { kapok_balance_political_power = 0 }
    }
    multiply_variable = { kapok_balance_political_power = 0.03 }  # 0-20转为0-0.6
    
    set_variable = { kapok_balance_civilian_bonus = 20 }
    subtract_from_variable = { kapok_balance_civilian_bonus = kapok_balance_penalty }
    if = {
        limit = { check_variable = { kapok_balance_civilian_bonus < 0 } }
        set_variable = { kapok_balance_civilian_bonus = 0 }
    }
    multiply_variable = { kapok_balance_civilian_bonus = 0.01 }  # 0-20转为0-0.2
    
    set_variable = { kapok_balance_military_bonus = kapok_balance_civilian_bonus }  # 军工奖励=民工奖励
}


# ==========================================
# 协同加成：两两指数联动红利
# 教育+产业 > 99 → 额外基础生产效率（上限+5%）
# 福利+基建 > 99 → 额外顺从度增长速度（上限+5%）
# 全指数均 > 49   → 触发结算事件（仅一次）
# ==========================================
LHS_calculate_synergy_bonus = {

    set_variable = { kapok_synergy_research_bonus = 0 }
    set_variable = { kapok_synergy_population_bonus = 0 }

    # --- 教育 × 产业 协同 ---
    set_variable = { kapok_temp_sum = kapok_education }
    add_to_variable = { kapok_temp_sum = kapok_industry }

    if = {
        limit = { check_variable = { kapok_temp_sum > 99 } }
        set_variable = { kapok_synergy_research_bonus = kapok_temp_sum }
        subtract_from_variable = { kapok_synergy_research_bonus = 100 }
        multiply_variable = { kapok_synergy_research_bonus = 0.001 }
        if = {
            limit = { check_variable = { kapok_synergy_research_bonus > 0.05 } }
            set_variable = { kapok_synergy_research_bonus = 0.05 }
        }
    }

    # --- 福利 × 基建 协同 ---
    set_variable = { kapok_temp_sum = kapok_welfare }
    add_to_variable = { kapok_temp_sum = kapok_infrastructure }

    if = {
        limit = { check_variable = { kapok_temp_sum > 99 } }
        set_variable = { kapok_synergy_population_bonus = kapok_temp_sum }
        subtract_from_variable = { kapok_synergy_population_bonus = 100 }
        multiply_variable = { kapok_synergy_population_bonus = 0.005 }
        if = {
            limit = { check_variable = { kapok_synergy_population_bonus > 0.05 } }
            set_variable = { kapok_synergy_population_bonus = 0.05 }
        }
    }

    # --- 全指数 > 49：事件检定（仅触发一次）---
    if = {
        limit = {
            check_variable = { kapok_infrastructure > 49 }
            check_variable = { kapok_welfare > 49 }
            check_variable = { kapok_education > 49 }
            check_variable = { kapok_industry > 49 }
            NOT = { has_country_flag = LHS_all_index_50_fired }
        }
        set_country_flag = LHS_all_index_50_fired
        country_event = { id = LHS.209 days = 1 }
    }
}
